// Safe division (na + zero protection)
safeDiv(float num, float den, float fallback = 0.0) =>
    den != 0 and not na(num) and not na(den) ? num / den : fallback

// ZigZag line style resolver

// Distribute candle volume across bins proportionally to candle range overlap
distributeVolume(float cOpen, float cClose, float cHigh, float cLow, float cVol, float bottom, float step, int bins, array<float> volArr, array<float> buyArr, array<float> sellArr) =>
    float candleTop = math.max(cHigh, cLow)
    float candleBot = math.min(cHigh, cLow)
    float candleRange = candleTop - candleBot
    bool isBullCandle = cClose >= cOpen
    if candleRange > 0 and cVol > 0
        for k = 0 to bins - 1
            float binLow  = bottom + k * step
            float binHigh = binLow + step
            float overlap = math.max(0.0, math.min(candleTop, binHigh) - math.max(candleBot, binLow))
            if overlap > 0
                float portion = overlap / candleRange
                float allocVol = cVol * portion
                volArr.set(k, volArr.get(k) + allocVol)
                if isBullCandle
                    buyArr.set(k, buyArr.get(k) + allocVol)
                else
                    sellArr.set(k, sellArr.get(k) + allocVol)
    true

// Calculate Value Area: expand from POC bin until VA_PERCENT of total volume captured
calcValueArea(array<float> volArr, float totalVol, int pocBin, float bottom, float step, int bins) =>
    float target = totalVol * VA_PERCENT
    float accumulated = volArr.get(pocBin)
    int upperBin = pocBin
    int lowerBin = pocBin
    int safety = 0
    while accumulated < target and safety < bins
        safety += 1
        float upVol   = upperBin + 1 < bins ? volArr.get(upperBin + 1) : 0.0
        float downVol = lowerBin - 1 >= 0   ? volArr.get(lowerBin - 1) : 0.0
        if upVol >= downVol and upperBin + 1 < bins
            upperBin += 1
            accumulated += upVol
        else if lowerBin - 1 >= 0
            lowerBin -= 1
            accumulated += downVol
        else if upperBin + 1 < bins
            upperBin += 1
            accumulated += upVol
        else
            break
    float vaHigh = bottom + (upperBin + 1) * step
    float vaLow  = bottom + lowerBin * step
    [vaHigh, vaLow]

        float step = swingRange / bins
        int lastIdx  = math.max(swingHighIdx, swingLowIdx)

        int barSpan  = math.max(lastIdx - prevIdx, 1)
        int offset   = bar_index - lastIdx

        array<float> buyBins  = array.new<float>(bins, 0.0)

        for i = offset to barSpan + offset

            float cC   = nz(close[i], 0.0)
            float cH   = na(high[i]) ? cC : high[i]
            float cL   = na(low[i]) ? cC : low[i]
            float cO   = na(open[i]) ? cC : open[i]
            distributeVolume(cO, cC, cH, cL, cVol, swingBot, step, bins, volBins, buyBins, sellBins)
            float typPrice = (cH + cL + cC) / 3.0
            sumPV += typPrice * cVol
            sumV  += cVol

        float totalVol = volBins.sum()
        float maxVol   = volBins.max()

        // Find POC bin
        int pocBin = 0
        if maxVol > 0
            for k = 0 to bins - 1
                if volBins.get(k) == maxVol
                    pocBin := k
                    break

        float pocPrice  = swingBot + pocBin * step + step / 2.0
        float pocBinLow = swingBot + pocBin * step
        float pocBinHi  = pocBinLow + step

        // Value Area
        float vaH = swingTop
        float vaL = swingBot
        if totalVol > 0
            [calcVaH, calcVaL] = calcValueArea(volBins, totalVol, pocBin, swingBot, step, bins)
            vaH := calcVaH
            vaL := calcVaL

        float swingVwap = safeDiv(sumPV, sumV, pocPrice)

        // Store for dashboard

        lastPocBinBot := pocBinLow
        lastVaHigh    := vaH

        int halfSpan = math.max(int(barSpan / 2), 1)

        for k = 0 to bins - 1
            float binL = swingBot + k * step
            float binH = binL + step
            float volRatio = safeDiv(volBins.get(k), maxVol, 0.0)
            int profWidth  = math.max(int(volRatio * halfSpan), 0)

            float buyV  = buyBins.get(k)
            float sellV = sellBins.get(k)
            color barColor = showDeltaInput ? (buyV >= sellV ? bullColorInput : bearColorInput) : swingColor

            // POC line — short segment inside profile
            if volRatio >= 0.99 and showPocInput
                line.new(showProfileInput ? prevIdx + profWidth : prevIdx, math.avg(binL, binH), lastIdx, math.avg(binL, binH), color = pocColorInput, width = pocWidthInput)

            if showProfileInput or showHeatmapInput

                string boxTxt = (not showHeatmapInput and volRatio >= 0.99) ? str.tostring(maxVol, format.volume) : ""
                box.new(showProfileInput ? prevIdx + profWidth : prevIdx, binH, showHeatmapInput ? lastIdx : prevIdx, binL, TRANSPARENT, 0, bgcolor = boxBg, text = boxTxt, text_color = TEXT_COLOR, text_size = size.tiny)

            polyPts.push(chart.point.from_index(prevIdx + profWidth, binH))
            if k == bins - 1

            label.new(prevIdx, dirIsDown ? swingTop : swingBot, lblText, style = dirIsDown ? label.style_label_down : label.style_label_up, color = TRANSPARENT, textcolor = swingColor, size = LABEL_SIZE, tooltip = tipText)

            vaH := calcVaH
            vaL := calcVaL

        float rtVwap = safeDiv(sumPV, sumV, pocPrice)

        // Update dashboard state
        lastPocPrice  := pocPrice
        lastVaHigh    := vaH
        lastVaLow     := vaL
        lastSwingVwap := rtVwap
        lastTotalVol  := totalVol
        lastBuyVol    := buyBins.sum()
        lastSellVol   := sellBins.sum()
        lastDelta     := safeDiv(lastBuyVol - lastSellVol, totalVol, 0.0) * 100.0

        array<chart.point> polyPts = array.new<chart.point>()
        int halfSpan = math.max(int(barSpan / 2), 1)

        for k = 0 to bins - 1
            float binL = rtBot + k * step
            float binH = binL + step
            float volRatio = safeDiv(volBins.get(k), maxVol, 0.0)
            int profWidth  = math.max(int(volRatio * halfSpan), 0)

            float buyV  = buyBins.get(k)
            float sellV = sellBins.get(k)
            color barColor = showDeltaInput ? (buyV >= sellV ? bullColorInput : bearColorInput) : rtColor

            // POC — delete before create (center + zone borders + fill)
            if volRatio >= 0.99 and showPocInput
                line.delete(rtPocLine)
                line.delete(rtPocTop)
                line.delete(rtPocBot)
                box.delete(rtPocFill)
                rtPocLine := line.new(showProfileInput ? prevIdx + profWidth : prevIdx, math.avg(binL, binH), lastIdx, math.avg(binL, binH), color = pocColorInput, width = pocWidthInput)
                rtPocTop  := line.new(showProfileInput ? prevIdx + profWidth : prevIdx, binH, lastIdx, binH, color = color.new(pocColorInput, 60), width = 1, style = line.style_dotted)
                rtPocBot  := line.new(showProfileInput ? prevIdx + profWidth : prevIdx, binL, lastIdx, binL, color = color.new(pocColorInput, 60), width = 1, style = line.style_dotted)
                if showPocFillInput
                    rtPocFill := box.new(showProfileInput ? prevIdx + profWidth : prevIdx, binH, lastIdx, binL, TRANSPARENT, 0, bgcolor = POC_FILL)

            // Box
            if showProfileInput or showHeatmapInput
                color boxBg = color.from_gradient(volRatio, 0.0, 1.0, color.new(barColor, 92), color.new(volRatio >= 0.99 and showPocInput ? pocColorInput : barColor, 50))
                string boxTxt = (not showHeatmapInput and volRatio >= 0.99) ? str.tostring(maxVol, format.volume) : ""
                rtBoxes.push(box.new(showProfileInput ? prevIdx + profWidth : prevIdx, binH, showHeatmapInput ? lastIdx : prevIdx, binL, TRANSPARENT, 0, bgcolor = boxBg, text = boxTxt, text_color = TEXT_COLOR, text_size = size.tiny))

            if k == 0
                polyPts.push(chart.point.from_index(prevIdx, binL))
            polyPts.push(chart.point.from_index(prevIdx + profWidth, binL))
            polyPts.push(chart.point.from_index(prevIdx + profWidth, binH))
            if k == bins - 1
                polyPts.push(chart.point.from_index(prevIdx, binH))

        if showProfileInput
            profilePoly := polyline.new(polyPts, false, true, line_color = showHeatmapInput ? color.new(rtColor, 50) : rtColor)

        // VA lines
        if showVaInput and totalVol > 0
            rtVaHighLine := line.new(prevIdx, vaH, lastIdx, vaH, color = vaColorInput, style = line.style_dashed, width = vaWidthInput)
            rtVaLowLine  := line.new(prevIdx, vaL, lastIdx, vaL, color = vaColorInput, style = line.style_dashed, width = vaWidthInput)

        // VWAP line
        if showVwapInput and totalVol > 0
            rtVwapLine := line.new(prevIdx, rtVwap, lastIdx, rtVwap, color = vwapColorInput, style = line.style_solid, width = 2)

        // Data label — dirIsDown: top label, not dirIsDown: bottom label
        string rtTip = "T - Total Volume\nB - Buy Volume\nS - Sell Volume\nD - Delta Volume"
        string rtDataText = "T: " + str.tostring(totalVol, format.volume) + "\nB: " + str.tostring(buyBins.sum(), format.volume) + "\nS: " + str.tostring(sellBins.sum(), format.volume) + "\nD: " + str.tostring(safeDiv(buyBins.sum() - sellBins.sum(), totalVol, 0.0) * 100.0, "#.0") + "%"

        if showLabelsInput
            rtLabel := label.new(prevIdx, dirIsDown ? rtTop : rtBot, rtDataText, style = dirIsDown ? label.style_label_down : label.style_label_up, color = TRANSPARENT, textcolor = rtColor, size = LABEL_SIZE, tooltip = rtTip, textalign = text.align_left)
